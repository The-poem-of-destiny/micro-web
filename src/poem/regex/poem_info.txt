<head>
<meta charset="utf-8">
<style>
  .pi{width:100%;margin:6px 0;font-size:14px;line-height:1.6}
  .pi-card{position:relative;border:1px solid var(--pi-accent-dim);border-left:3px solid var(--pi-accent);border-radius:8px;background:rgba(15,17,26,.92);color:#e8e3d5;overflow:hidden}
  .pi-card::before{content:'';position:absolute;inset:0;border-radius:8px;background:linear-gradient(135deg,var(--pi-accent),transparent 40%);opacity:.05;pointer-events:none}
  .pi-head{display:flex;align-items:center;gap:8px;padding:7px 12px;border-bottom:1px solid var(--pi-accent-dim);background:linear-gradient(90deg,var(--pi-accent-bg),transparent 70%);font-size:14px;font-weight:600;letter-spacing:2px;color:var(--pi-accent-strong)}
  .pi-head-ico{font-weight:400}
  .pi-head-tag{margin-left:auto;padding:0 10px;border:1px solid var(--pi-accent-dim);border-radius:999px;font-size:12px;font-weight:400;letter-spacing:1px;color:var(--pi-accent)}
  .pi-body{padding:8px 12px 10px}
  .pi-row{display:flex;gap:8px;padding:1px 0;align-items:baseline}
  .pi-k{flex:none;min-width:3em;color:rgba(232,227,213,.5);font-size:12px}
  .pi-v{flex:1;min-width:0;word-break:break-word}
  .pi-sub{margin:2px 0 2px;padding-left:8px;border-left:1px solid var(--pi-accent-dim)}
  .pi-item{padding:1px 0;color:rgba(232,227,213,.85);font-size:13px}
  .pi-q{display:inline-block;padding:0 8px;border:1px solid;border-radius:999px;font-size:12px}
  .pi-pill{display:inline-block;margin:1px 2px 1px 0;padding:0 8px;border-radius:999px;background:rgba(232,227,213,.08);font-size:12px}
  .pi a{color:var(--pi-accent);text-decoration:none;cursor:pointer;white-space:nowrap}

<div class="pi" data-kind="$1"><pre class="pi-src">$2</pre></div>
<script>
(function () {
  var KIND = {
    battle: { title: '战斗', ico: '⚔', accent: '#c4574e' },
    item: { title: '物品', ico: '◆', accent: '#c8b478' },
    skill: { title: '技能', ico: '✦', accent: '#7aa2c4' },
    task: { title: '任务', ico: '❖', accent: '#7cb47f' },
    action: { title: '行动', ico: '➤', accent: '#9b8fb0' },
    asset: { title: '资产', ico: '⌂', accent: '#b08d6e' },
    buff: { title: '状态', ico: '◈', accent: '#a97fc0' },
    mfg: { title: '制作', ico: '⚒', accent: '#c4884f' },
    char: { title: '角色', ico: '☾', accent: '#6fb3a8' }
  };
  var QUALITY = {
    '唯一': '#e8c15a', '神话': '#e05c5c', '传说': '#a05cc4', '史诗': '#e08a3c',
    '稀有': '#4c9be0', '优良': '#4cc47e', '普通': '#9aa5b1'
  };

  function shade(hex, k) {
    var n = parseInt(hex.slice(1), 16);
    return 'rgb(' + Math.round(((n >> 16) & 255) * k) + ',' + Math.round(((n >> 8) & 255) * k) + ',' + Math.round((n & 255) * k) + ')';
  }

  // 长文：截断 + inline 展开/收起
  function longText(text, limit) {
    if (text.length <= limit) { var s = document.createElement('span'); s.textContent = text; return s; }
    var box = document.createElement('span');
    var head = document.createElement('span'); head.textContent = text.slice(0, limit) + '……';
    var rest = document.createElement('span'); rest.style.display = 'none'; rest.textContent = text.slice(limit);
    var btn = document.createElement('a'); btn.href = 'javascript:void(0)'; btn.textContent = '展开';
    // DOM 顺序 head+rest+btn：收起态 rest 隐藏，btn 贴省略号后；展开态 btn 落全文末尾
    btn.onclick = function () {
      var open = rest.style.display !== 'none';
      rest.style.display = open ? 'none' : '';
      head.textContent = text.slice(0, limit) + (open ? '……' : '');
      btn.textContent = open ? '展开' : '收起';
    };
    box.append(head, rest, btn);
    return box;
  }

  // 字符串值：品质 chip → | pill → 长文
  function inlineValue(v) {
    var q = QUALITY[v];
    if (q) {
      var chip = document.createElement('span');
      chip.className = 'pi-q'; chip.textContent = v;
      chip.style.borderColor = q; chip.style.color = q;
      return chip;
    }
    if (v.charAt(0) === '[' && v.indexOf(' | ') !== -1 && v.length <= 120) {
      var box = document.createElement('span');
      v.split(' | ').forEach(function (seg) {
        var pill = document.createElement('span');
        pill.className = 'pi-pill'; pill.textContent = seg.trim();
        box.append(pill);
      });
      return box;
    }
    var span = document.createElement('span');
    span.className = 'pi-v';
    span.append(longText(v, 80));
    return span;
  }

  // 嵌套对象 → 键值组；数组 → 条目组；字符串 → 行内
  function renderAny(val) {
    if (val === null || val === undefined) return document.createTextNode('');
    if (typeof val === 'object' && !Array.isArray(val)) {
      var sub = document.createElement('div');
      sub.className = 'pi-sub';
      Object.keys(val).forEach(function (k) {
        var row = document.createElement('div');
        row.className = 'pi-row';
        var kk = document.createElement('span');
        kk.className = 'pi-k'; kk.textContent = k;
        var vv = document.createElement('span');
        vv.className = 'pi-v';
        vv.append(renderAny(val[k]));
        row.append(kk, vv);
        sub.append(row);
      });
      return sub;
    }
    if (Array.isArray(val)) {
      var list = document.createElement('div');
      list.className = 'pi-sub';
      val.forEach(function (item) {
        var li = document.createElement('div');
        li.className = 'pi-item';
        if (typeof item === 'string') li.append(inlineValue(item));
        else li.append(renderAny(item));
        list.append(li);
      });
      return list;
    }
    return inlineValue(String(val));
  }

  // 顶层对象 → 键值行
  function renderMap(map) {
    var box = document.createElement('div');
    Object.keys(map).forEach(function (k) {
      var row = document.createElement('div');
      row.className = 'pi-row';
      var kk = document.createElement('span');
      kk.className = 'pi-k'; kk.textContent = k;
      var vv = document.createElement('span');
      vv.className = 'pi-v';
      var val = map[k];
      if (val !== null && typeof val === 'object') vv.append(renderAny(val));
      else vv.append(inlineValue(String(val)));
      row.append(kk, vv);
      box.append(row);
    });
    return box;
  }

  // 兜底：非严格 YAML 时的逐行渲染（保留原缩进层级，2 空格为一层）
  function renderLines(rawLines) {
    var box = document.createElement('div');
    var base = Infinity;
    rawLines.forEach(function (l) {
      if (l.trim() === '') return;
      var ind = l.match(/^\s*/)[0].length;
      if (ind < base) base = ind;
    });
    if (!isFinite(base)) base = 0;
    rawLines.forEach(function (l) {
      var t = l.trim();
      if (t === '') return;
      var depth = Math.round((l.match(/^\s*/)[0].length - base) / 2);
      var m = t.match(/^(-?)\s*"?([^:：]+)"?[:：]\s*(.*)$/);
      var row = document.createElement('div');
      if (m && m[2].trim() !== '') {
        row.className = 'pi-row';
        row.style.marginLeft = depth * 1.2 + 'em';
        var k = document.createElement('span');
        k.className = 'pi-k'; k.textContent = m[2].trim().replace(/"/g, '');
        var v = document.createElement('span');
        v.className = 'pi-v';
        v.append(inlineValue(m[3].trim().replace(/^"|"$/g, '')));
        row.append(k, v);
      } else {
        row.className = 'pi-item';
        row.style.marginLeft = depth * 1.2 + 'em';
        row.textContent = t.replace(/^-\s*/, '');
      }
      box.append(row);
    });
    return box;
  }

  function build(root, yamlLib) {
    var kindKey = root.getAttribute('data-kind');
    var kind = KIND[kindKey] || { title: '信息', ico: '◆', accent: '#9aa5b1' };
    var text = (root.querySelector('.pi-src').textContent || '').trim();

    // 头部右侧子阶段徽标
    var tag = '';
    var mm = /阶段[:：]\s*(\S+)/.exec(text);
    var mc = /交锋[:：]\s*"?第?(\d+)次?"?/.exec(text);
    if (kindKey === 'battle') {
      if (mm) tag = mm[1];
      else if (mc) tag = '交锋 · 第' + mc[1] + '次';
    } else if (mm) {
      tag = mm[1];
    }

    var card = document.createElement('div');
    card.className = 'pi-card';
    card.style.setProperty('--pi-accent', kind.accent);
    card.style.setProperty('--pi-accent-dim', shade(kind.accent, 0.45));
    card.style.setProperty('--pi-accent-bg', 'rgba(' + shade(kind.accent, 0.16).slice(4));

    var head = document.createElement('div');
    head.className = 'pi-head';
    var ico = document.createElement('span');
    ico.className = 'pi-head-ico'; ico.textContent = kind.ico;
    var title = document.createElement('span');
    title.textContent = kind.title;
    head.append(ico, title);
    if (tag) {
      var tagEl = document.createElement('span');
      tagEl.className = 'pi-head-tag'; tagEl.textContent = tag;
      head.append(tagEl);
    }

    var body = null;
    if (yamlLib) {
      try { body = renderMap(yamlLib.parse(text)); } catch (e) { body = null; }
    }
    if (!body) {
      body = renderLines(text.split('\n'));
    }

    var bodyBox = document.createElement('div');
    bodyBox.className = 'pi-body';
    bodyBox.append(body);

    card.append(head, bodyBox);
    root.textContent = '';
    root.append(card);
  }

  function boot(root) {
    // 优先复用酒馆环境全局 YAML，否则动态加载 js-yaml CDN
    var y = (window.parent && window.parent.YAML) || window.YAML;
    if (y && y.parse) { build(root, y); return; }
    var s = document.createElement('script');
    s.src = 'https://testingcf.jsdelivr.net/npm/js-yaml@4.1.0/dist/js-yaml.min.js';
    s.onload = function () { build(root, { parse: function (t) { return window.jsyaml.load(t); } }); };
    s.onerror = function () { build(root, null); };
    document.head.append(s);
  }

  var root = document.querySelector('.pi');
  if (root) boot(root);
})();
</script>
</body>
